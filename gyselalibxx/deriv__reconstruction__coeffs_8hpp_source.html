// SPDX-License-Identifier: MIT

#pragma once

#include <ddc/ddc.hpp>

#include "ddc_aliases.hpp"
#include "geometry_descriptors.hpp"
#include "math_tools.hpp"


template <
        class EdgePerpGrid1Type,
        class EdgePerpGrid2Type,
        class BSplinesPerp1Type,
        class BSplinesPerp2Type,
        Extremity extremity_1,
        Extremity extremity_2>
class DerivReconstructionCoeffs
{
protected:
    using GridEdgePerp1 = EdgePerpGrid1Type;
    using GridEdgePerp2 = EdgePerpGrid2Type;

    using IdxRange1DPerp_1 = IdxRange<GridEdgePerp1>;
    using IdxRange1DPerp_2 = IdxRange<GridEdgePerp2>;

    using IdxEdgePerp1 = Idx<GridEdgePerp1>;
    using IdxEdgePerp2 = Idx<GridEdgePerp2>;

    using BSplinesPerp1 = BSplinesPerp1Type;
    using BSplinesPerp2 = BSplinesPerp2Type;

    using GridBreakPt1 = ddc::knot_discrete_dimension_t<BSplinesPerp1>;
    using GridBreakPt2 = ddc::knot_discrete_dimension_t<BSplinesPerp2>;

protected:
    const bool m_is_cell_bound_1_with_extra_interpol_pt;
    const bool m_is_cell_bound_2_with_extra_interpol_pt;

    IdxRange1DPerp_1 const m_idx_range_perp_1;
    IdxRange1DPerp_2 const m_idx_range_perp_2;

    static Extremity constexpr s_extremity_1 = extremity_1;
    static Extremity constexpr s_extremity_2 = extremity_2;

    static constexpr std::size_t n = 2;
    static constexpr std::size_t n_minus_1 = 1;
    static constexpr std::size_t n_minus_2 = 0;

    double m_coeff_deriv_patch_1;
    double m_coeff_deriv_patch_2;

    host_t<DFieldMem<IdxRange1DPerp_1>> m_weights_patch_1_alloc;
    host_t<DFieldMem<IdxRange1DPerp_2>> m_weights_patch_2_alloc;

    host_t<DField<IdxRange1DPerp_1>> m_weights_patch_1;
    host_t<DField<IdxRange1DPerp_2>> m_weights_patch_2;

public:
    DerivReconstructionCoeffs(
            IdxRange1DPerp_1 const& idx_range_1d_1,
            IdxRange1DPerp_2 const& idx_range_1d_2,
            ddc::SplineBuilderClosure const& Closure1 = ddc::SplineBuilderClosure::HERMITE,
            ddc::SplineBuilderClosure const& Closure2 = ddc::SplineBuilderClosure::HERMITE)
        : m_is_cell_bound_1_with_extra_interpol_pt(Closure1 == ddc::SplineBuilderClosure::GREVILLE)
        , m_is_cell_bound_2_with_extra_interpol_pt(Closure2 == ddc::SplineBuilderClosure::GREVILLE)
        , m_idx_range_perp_1(idx_range_1d_1)
        , m_idx_range_perp_2(idx_range_1d_2)
        , m_weights_patch_1_alloc(
                  "m_weights_patch_1 "
                  "(DerivReconstructionCoeffs::DerivReconstructionCoeffs)",
                  m_idx_range_perp_1)
        , m_weights_patch_2_alloc(
                  "m_weights_patch_2 "
                  "(DerivReconstructionCoeffs::DerivReconstructionCoeffs)",
                  m_idx_range_perp_2)
        , m_weights_patch_1(m_weights_patch_1_alloc)
        , m_weights_patch_2(m_weights_patch_2_alloc)
    {
        assert(Closure1 != ddc::SplineBuilderClosure::PERIODIC);
        assert(Closure2 != ddc::SplineBuilderClosure::PERIODIC);

        // Two interpolation points have to be added if the derivatives are not closure condition.
        if (m_is_cell_bound_1_with_extra_interpol_pt) {
            assert(m_idx_range_perp_1.size() == ddc::discrete_space<BSplinesPerp1>().ncells() + 2);
        } else {
            assert(m_idx_range_perp_1.size() <= ddc::discrete_space<BSplinesPerp1>().ncells() + 1);
        }
        if (m_is_cell_bound_2_with_extra_interpol_pt) {
            assert(m_idx_range_perp_2.size() == ddc::discrete_space<BSplinesPerp2>().ncells() + 2);
        } else {
            assert(m_idx_range_perp_2.size() <= ddc::discrete_space<BSplinesPerp2>().ncells() + 1);
        }

        // The break points have to be interpolation points.
        check_break_points_are_interpolation_points<BSplinesPerp1, GridBreakPt1>(
                m_is_cell_bound_1_with_extra_interpol_pt,
                m_idx_range_perp_1,
                s_extremity_1);
        check_break_points_are_interpolation_points<BSplinesPerp2, GridBreakPt2>(
                m_is_cell_bound_2_with_extra_interpol_pt,
                m_idx_range_perp_2,
                s_extremity_2);


        // The additional interpolation points have to be in the boundary cells.
        if (m_is_cell_bound_1_with_extra_interpol_pt) {
            check_additional_interpolation_points_location<
                    BSplinesPerp1,
                    GridBreakPt1>(m_idx_range_perp_1, s_extremity_1);
        }
        if (m_is_cell_bound_2_with_extra_interpol_pt) {
            check_additional_interpolation_points_location<
                    BSplinesPerp2,
                    GridBreakPt2>(m_idx_range_perp_2, s_extremity_2);
        }

        if constexpr ((ddc::is_uniform_point_sampling_v<GridEdgePerp1>)&&(
                              ddc::is_uniform_point_sampling_v<GridEdgePerp2>)) {
            set_coefficients_uniform_per_patch_case();
        } else {
            set_coefficients_non_uniform_case();
        }
    }


    template <class IdxRangeA, class IdxRangeB>
    DerivReconstructionCoeffs(
            IdxRangeA const& idx_range_a,
            IdxRangeB const& idx_range_b,
            ddc::SplineBuilderClosure const& Closure1 = ddc::SplineBuilderClosure::HERMITE,
            ddc::SplineBuilderClosure const& Closure2 = ddc::SplineBuilderClosure::HERMITE)
        : DerivReconstructionCoeffs(
                IdxRange1DPerp_1(idx_range_a, idx_range_b),
                IdxRange1DPerp_2(idx_range_a, idx_range_b),
                Closure1,
                Closure2)
    {
        static_assert(ddc::is_discrete_domain_v<IdxRangeA>);
        static_assert(ddc::is_discrete_domain_v<IdxRangeB>);
    }

    DerivReconstructionCoeffs(
            IdxRange1DPerp_1 const& idx_range_1d_1,
            IdxRange1DPerp_2 const& idx_range_1d_2,
            std::size_t const number_chosen_cells)
        : DerivReconstructionCoeffs(
                (s_extremity_1 == Extremity::FRONT)
                        ? idx_range_1d_1.take_first(IdxStep<GridEdgePerp1>(
                                Kokkos::min(number_chosen_cells + 1, idx_range_1d_1.size())))
                        : idx_range_1d_1.take_last(IdxStep<GridEdgePerp1>(
                                Kokkos::min(number_chosen_cells + 1, idx_range_1d_1.size()))),
                (s_extremity_2 == Extremity::FRONT)
                        ? idx_range_1d_2.take_first(IdxStep<GridEdgePerp2>(
                                Kokkos::min(number_chosen_cells + 1, idx_range_1d_2.size())))
                        : idx_range_1d_2.take_last(IdxStep<GridEdgePerp2>(
                                Kokkos::min(number_chosen_cells + 1, idx_range_1d_2.size()))),
                ddc::SplineBuilderClosure::HERMITE,
                ddc::SplineBuilderClosure::HERMITE)
    {
    }

    template <class IdxRangeA, class IdxRangeB>
    DerivReconstructionCoeffs(
            IdxRangeA const& idx_range_a,
            IdxRangeB const& idx_range_b,
            std::size_t const number_chosen_cells) requires(!std::
                                                                    is_same_v<
                                                                            IdxRange1DPerp_1,
                                                                            IdxRange1DPerp_2>)
        : DerivReconstructionCoeffs(
                IdxRange1DPerp_1(idx_range_a, idx_range_b),
                IdxRange1DPerp_2(idx_range_a, idx_range_b),
                number_chosen_cells)
    {
        static_assert(ddc::is_discrete_domain_v<IdxRangeA>);
        static_assert(ddc::is_discrete_domain_v<IdxRangeB>);
    }

    double get_coeff_deriv_patch_1() const
    {
        return m_coeff_deriv_patch_1;
    }

    double get_coeff_deriv_patch_2() const
    {
        return m_coeff_deriv_patch_2;
    }

    template <class Layout1, class Layout2>
    inline double get_approx_deriv(
            DConstField<IdxRange1DPerp_1, Kokkos::HostSpace, Layout1> const& function_1,
            DConstField<IdxRange1DPerp_2, Kokkos::HostSpace, Layout2> const& function_2) const
    {
        return get_approx_with_change_of_sign(function_1, function_2, 1);
    }

    template <class Layout1, class Layout2>
    inline double get_approx_deriv(
            DConstField<IdxRange1DPerp_2, Kokkos::HostSpace, Layout2> const& function_2,
            DConstField<IdxRange1DPerp_1, Kokkos::HostSpace, Layout1> const& function_1) const
            requires(!std::is_same_v<IdxRange1DPerp_2, IdxRange1DPerp_1>)
    {
        return get_approx_with_change_of_sign(function_1, function_2, 1);
    }

    template <class Layout1, class Layout2>
    inline double get_approx_cross_deriv(
            DConstField<IdxRange1DPerp_1, Kokkos::HostSpace, Layout1> const& derivs_1,
            DConstField<IdxRange1DPerp_2, Kokkos::HostSpace, Layout2> const& derivs_2,
            int const sign_deriv_2) const
    {
        return this->get_approx_with_change_of_sign(derivs_1, derivs_2, sign_deriv_2);
    }

protected:
    template <class Layout1, class Layout2>
    double get_approx_with_change_of_sign(
            DConstField<IdxRange1DPerp_1, Kokkos::HostSpace, Layout1> const& function_1,
            DConstField<IdxRange1DPerp_2, Kokkos::HostSpace, Layout2> const& function_2,
            int const sign_function_2) const
    {
        // The function needs to be continuous at the interface.
        assert(abs(function_1(get_extremity_idx(s_extremity_1, m_idx_range_perp_1))
                   - sign_function_2
                             * function_2(get_extremity_idx(s_extremity_2, m_idx_range_perp_2)))
               < 1e-13);

        double coeff_values = ddc::host_transform_reduce(
                m_idx_range_perp_1,
                0.0,
                ddc::reducer::sum<double>(),
                [&](IdxEdgePerp1 const& idx) { return function_1(idx) * m_weights_patch_1(idx); });

        // To avoid counting twice the value at the interface.
        coeff_values += ddc::host_transform_reduce(
                m_idx_range_perp_2,
                0.0,
                ddc::reducer::sum<double>(),
                [&](IdxEdgePerp2 const& idx) {
                    return sign_function_2 * function_2(idx) * m_weights_patch_2(idx);
                });
        return coeff_values;
    }

    template <typename BSplinesPerp, typename GridBreakPt, typename EdgePerpGrid>
    void check_break_points_are_interpolation_points(
            bool const is_cell_bound_with_extra_interpol_pt,
            IdxRange<EdgePerpGrid> const& idx_range_perp,
            Extremity const extremity) const
    {
        // Select the subset of break points where the interpolation points must be defined.
        IdxRange<GridBreakPt> idx_range_break_points
                = ddc::discrete_space<BSplinesPerp>().break_point_domain();
        IdxStep<GridBreakPt> idx_step(idx_range_perp.size() - is_cell_bound_with_extra_interpol_pt);
        if (extremity == Extremity::FRONT) {
            idx_range_break_points = idx_range_break_points.take_first(idx_step);
        } else {
            idx_range_break_points = idx_range_break_points.take_last(idx_step);
        }
        Idx<GridBreakPt> missing_breakpoint = ddcHelper::
                find_first_point_not_in_superset(idx_range_break_points, idx_range_perp);
        if (missing_breakpoint != (idx_range_break_points.back() + 1)) {
            throw std::runtime_error(
                    "[abort] The break points have to be interpolation points. The break point "
                    + std::to_string(ddc::coordinate(missing_breakpoint)) + " at the "
                    + std::to_string((missing_breakpoint - idx_range_break_points.front()).value())
                    + "th index was not found in the interpolation point grid given.");
        }
    }

    template <typename BSplinesPerp, typename GridBreakPt, typename EdgePerpGrid>
    void check_additional_interpolation_points_location(
            IdxRange<EdgePerpGrid> const& idx_range_perp,
            Extremity const extremity)
    {
        IdxRange<GridBreakPt> idx_range_full_break_points
                = ddc::discrete_space<BSplinesPerp>().break_point_domain();

        double added_interpolation_pt;
        double break_coord_min;
        double break_coord_max;
        if (extremity == Extremity::FRONT) {
            Idx<EdgePerpGrid> idx = idx_range_perp.back() - IdxStep<EdgePerpGrid>(1);
            Idx<GridBreakPt> idx_break
                    = idx_range_full_break_points.back() - IdxStep<GridBreakPt>(1);
            added_interpolation_pt = double(ddc::coordinate(idx));
            break_coord_min = double(ddc::coordinate(idx_break));
            break_coord_max = double(ddc::coordinate(idx_range_full_break_points.back()));
        } else {
            Idx<EdgePerpGrid> idx = idx_range_perp.front() + IdxStep<EdgePerpGrid>(1);
            Idx<GridBreakPt> idx_break
                    = idx_range_full_break_points.front() + IdxStep<GridBreakPt>(1);
            added_interpolation_pt = double(ddc::coordinate(idx));
            break_coord_min = double(ddc::coordinate(idx_range_full_break_points.front()));
            break_coord_max = double(ddc::coordinate(idx_break));
        }

        if (!((break_coord_min < added_interpolation_pt)
              && (added_interpolation_pt < break_coord_max))) {
            throw std::runtime_error(
                    "[abort] The additional interpolation points have to be placed in the "
                    "first or last cell of the patch. The point "
                    + std::to_string(added_interpolation_pt) + " is not placed between "
                    + std::to_string(break_coord_min) + " and " + std::to_string(break_coord_max)
                    + ".");
        }
    }

    // ===========================================================================================
    // NON-UNIFORM INTERPOLATION POINTS                                                          |
    // ===========================================================================================
    void set_coefficients_non_uniform_case()
    {
        // Memory allocation ---------------------------------------------------------------------
        // Define weight fields for number of cells equal to n, n-1 and n-2 in the recursion.
        host_t<DFieldMem<IdxRange1DPerp_1>> weights_1_buffer_1_alloc(m_idx_range_perp_1);
        host_t<DFieldMem<IdxRange1DPerp_1>> weights_1_buffer_2_alloc(m_idx_range_perp_1);

        host_t<DFieldMem<IdxRange1DPerp_2>> weights_2_buffer_1_alloc(m_idx_range_perp_2);
        host_t<DFieldMem<IdxRange1DPerp_2>> weights_2_buffer_2_alloc(m_idx_range_perp_2);

        int const n_points_1 = m_idx_range_perp_1.size();
        int const n_points_2 = m_idx_range_perp_2.size();

        // Please provide at least 2 cells.
        if (n_points_1 < 2) {
            throw std::runtime_error("Please provide at least 2 cells in the Patch1.");
        }
        if (n_points_2 < 2) {
            throw std::runtime_error("Please provide at least 2 cells in the Patch2.");
        }

        // If m_is_cell_bound_1_with_extra_interpol_pt is true, the last cell is treated differently.
        int const n1_cells
                = m_is_cell_bound_1_with_extra_interpol_pt ? n_points_1 - 3 : n_points_1 - 1;
        // If m_is_cell_bound_2_with_extra_interpol_pt is true, the last cell is treated differently.
        int const n2_cells
                = m_is_cell_bound_2_with_extra_interpol_pt ? n_points_2 - 3 : n_points_2 - 1;

        // The number of times the buffers are cycled
        int n_permutations = (n1_cells - 2) + (n2_cells - 2)
                             + m_is_cell_bound_2_with_extra_interpol_pt
                             + m_is_cell_bound_1_with_extra_interpol_pt + 1;
        // The position of m_weights_patch_1/m_weights_patch_2 in the array weights_patch1/weights_patch2
        // so it is found at index n=2 at the end of the calculations
        int input_position = (n_permutations + 2) % 3;

        // Store the 3 last steps for the recursion in an array.
        std::array<host_t<DField<IdxRange1DPerp_1>>, 3> weights_patch1 = build_array(
                input_position,
                m_weights_patch_1,
                get_field(weights_1_buffer_1_alloc),
                get_field(weights_1_buffer_2_alloc));
        std::array<host_t<DField<IdxRange1DPerp_2>>, 3> weights_patch2 = build_array(
                input_position,
                m_weights_patch_2,
                get_field(weights_2_buffer_1_alloc),
                get_field(weights_2_buffer_2_alloc));

        // Initialise fields
        ddc::parallel_fill(weights_patch1[n], 0.0);
        ddc::parallel_fill(weights_patch1[n_minus_1], 0.0);
        ddc::parallel_fill(weights_patch1[n_minus_2], 0.0);
        ddc::parallel_fill(weights_patch2[n], 0.0);
        ddc::parallel_fill(weights_patch2[n_minus_1], 0.0);
        ddc::parallel_fill(weights_patch2[n_minus_2], 0.0);

        std::array<double, 3> coeff_deriv_patch1;
        std::array<double, 3> coeff_deriv_patch2;

        // For the local coefficients
        double alpha_i;
        double beta_i;
        std::array<double, 3> gammas_i;

        // Define a step to increment the indices to move away from the interface.
        IdxStep<GridEdgePerp1> idx_step_1(s_extremity_1 == FRONT ? 1 : -1);
        IdxStep<GridEdgePerp2> idx_step_2(s_extremity_2 == FRONT ? 1 : -1);

        // Initialisation recursivity: set (a_{n-2},b_{n-2},c_{n-2}) and (a_{n-1},b_{n-1},c_{n-1}) -----------------------------
        IdxEdgePerp1 interface_idx_1 = get_extremity_idx(s_extremity_1, m_idx_range_perp_1);
        IdxEdgePerp2 interface_idx_2 = get_extremity_idx(s_extremity_2, m_idx_range_perp_2);

        // Set (a_{n-2},b_{n-2},c_{n-2}) ---
        IdxEdgePerp1 idx_1_minus;
        IdxEdgePerp1 idx_1 = interface_idx_1;
        IdxEdgePerp1 idx_1_plus = interface_idx_1 + idx_step_1;

        IdxEdgePerp2 idx_2_minus;
        IdxEdgePerp2 idx_2 = interface_idx_2;
        IdxEdgePerp2 idx_2_plus = interface_idx_2 + idx_step_2;

        auto [cell_length_left, cell_length_right] = get_cell_lengths<false>(idx_2);

        alpha_i = get_alpha(cell_length_left, cell_length_right);
        beta_i = get_beta(cell_length_left, cell_length_right);
        gammas_i = get_gammas(cell_length_left, cell_length_right);

        coeff_deriv_patch1[n_minus_1] = beta_i; // b_{n-1}
        coeff_deriv_patch2[n_minus_1] = alpha_i; // a_{n-1}

        // c_{n-1}
        weights_patch1[n_minus_1](idx_1_plus) = gammas_i[0];
        weights_patch1[n_minus_1](idx_1) = gammas_i[1];
        weights_patch2[n_minus_1](idx_2) = gammas_i[1];
        weights_patch2[n_minus_1](idx_2_plus) = gammas_i[2];

        // Set (a_{n},b_{n},c_{n}) ---
        idx_2_minus = interface_idx_2;
        idx_2 = interface_idx_2 + idx_step_2;
        idx_2_plus = interface_idx_2 + 2 * idx_step_2;

        std::tie(cell_length_left, cell_length_right) = get_cell_lengths<false>(idx_2);

        alpha_i = get_alpha(cell_length_left, cell_length_right);
        beta_i = get_beta(cell_length_left, cell_length_right);
        gammas_i = get_gammas(cell_length_left, cell_length_right);

        coeff_deriv_patch2[n] = alpha_i * coeff_deriv_patch2[n_minus_1]
                                / (1 - coeff_deriv_patch2[n_minus_1] * beta_i); // a_{n}
        coeff_deriv_patch1[n] = coeff_deriv_patch1[n_minus_1]
                                / (1 - coeff_deriv_patch2[n_minus_1] * beta_i); // b_{n}

        // c_{n} = (c_{n-1} + a_{n-1} * gamma_i) / (1 - a_{n-1} * beta_i);
        ddc::host_for_each(m_idx_range_perp_1, [&](IdxEdgePerp1 const& idx) {
            weights_patch1[n](idx) = 1. / (1 - coeff_deriv_patch2[n_minus_1] * beta_i)
                                     * weights_patch1[n_minus_1](idx);
        });
        ddc::host_for_each(m_idx_range_perp_2, [&](IdxEdgePerp2 const& idx) {
            weights_patch2[n](idx) = 1. / (1 - coeff_deriv_patch2[n_minus_1] * beta_i)
                                     * weights_patch2[n_minus_1](idx);
        });
        double factor
                = 1. / (1 - coeff_deriv_patch2[n_minus_1] * beta_i) * coeff_deriv_patch2[n_minus_1];
        weights_patch2[n](idx_2_plus) += factor * gammas_i[2];
        weights_patch2[n](idx_2) += factor * gammas_i[1];
        weights_patch2[n](idx_2_minus) += factor * gammas_i[0];
        weights_patch1[n](idx_1) += factor * gammas_i[0];


        // Computing coefficients (c, a, b) on patch 2 -------------------------------------------
        recursion<false>(
                n2_cells,
                interface_idx_2,
                coeff_deriv_patch2,
                coeff_deriv_patch1,
                weights_patch1,
                weights_patch2);

        // Correction if we use an additional interpolation point as closure in the last cell.
        if (m_is_cell_bound_2_with_extra_interpol_pt) {
            correction_boundary<false>(
                    n_points_2,
                    interface_idx_2,
                    coeff_deriv_patch2,
                    coeff_deriv_patch1,
                    weights_patch1,
                    weights_patch2);
        }

        cycle_buffers(coeff_deriv_patch1);
        cycle_buffers(coeff_deriv_patch2);
        cycle_buffers(weights_patch1);
        cycle_buffers(weights_patch2);

        // Initialisation second recursivity: set (a_{n-2},b_{n-2},c_{n-2}) and (a_{n-1},b_{n-1},c_{n-1}) ----------------------
        // Set (a_{n-2},b_{n-2},c_{n-2}) ---
        idx_1_minus = interface_idx_1;
        idx_1 = interface_idx_1 + idx_step_1;
        idx_1_plus = interface_idx_1 + 2 * idx_step_1;

        std::tie(cell_length_left, cell_length_right) = get_cell_lengths<true>(idx_1);

        alpha_i = get_alpha(cell_length_left, cell_length_right);
        beta_i = get_beta(cell_length_left, cell_length_right);
        gammas_i = get_gammas(cell_length_left, cell_length_right);

        // Set (a_{n},b_{n},c_{n}) ---
        coeff_deriv_patch2[n] = coeff_deriv_patch2[n_minus_1]
                                / (1 - coeff_deriv_patch1[n_minus_1] * alpha_i); // a_{n}
        coeff_deriv_patch1[n] = coeff_deriv_patch1[n_minus_1] * beta_i
                                / (1 - coeff_deriv_patch1[n_minus_1] * alpha_i); // b_{n}

        // c_{n} = (c_{n-1} + b_{n-1} * gamma_i) / (1 - b_{n-1} * alpha_i);
        ddc::host_for_each(m_idx_range_perp_1, [&](IdxEdgePerp1 const& idx) {
            weights_patch1[n](idx) = 1. / (1 - coeff_deriv_patch1[n_minus_1] * alpha_i)
                                     * weights_patch1[n_minus_1](idx);
        });
        ddc::host_for_each(m_idx_range_perp_2, [&](IdxEdgePerp2 const& idx) {
            weights_patch2[n](idx) = 1. / (1 - coeff_deriv_patch1[n_minus_1] * alpha_i)
                                     * weights_patch2[n_minus_1](idx);
        });
        factor = 1. / (1 - coeff_deriv_patch1[n_minus_1] * alpha_i) * coeff_deriv_patch1[n_minus_1];
        weights_patch1[n](idx_1_plus) += factor * gammas_i[0];
        weights_patch1[n](idx_1) += factor * gammas_i[1];
        weights_patch1[n](idx_1_minus) += factor * gammas_i[2];
        weights_patch2[n](interface_idx_2) += factor * gammas_i[2];

        // Computing coefficients (c, a, b) on patch 1 -------------------------------------------
        recursion<true>(
                n1_cells,
                interface_idx_1,
                coeff_deriv_patch1,
                coeff_deriv_patch2,
                weights_patch1,
                weights_patch2);

        // Correction if we use an additional interpolation point as closure in the last cell.
        if (m_is_cell_bound_1_with_extra_interpol_pt) {
            correction_boundary<true>(
                    n_points_1,
                    interface_idx_1,
                    coeff_deriv_patch1,
                    coeff_deriv_patch2,
                    weights_patch1,
                    weights_patch2);
        }

        // Store the final values.
        m_coeff_deriv_patch_1 = coeff_deriv_patch1[n];
        m_coeff_deriv_patch_2 = coeff_deriv_patch2[n];
        m_weights_patch_1(interface_idx_1) *= 0.5;
        m_weights_patch_2(interface_idx_2) *= 0.5;
    }


    template <bool is_on_patch1, class Idx1D>
    void recursion(
            int const n_cells,
            Idx1D const interface_idx,
            std::array<double, 3>& coeff_deriv_patch,
            std::array<double, 3>& coeff_deriv_other_patch,
            std::array<host_t<DField<IdxRange1DPerp_1>>, 3>& weights_patch1,
            std::array<host_t<DField<IdxRange1DPerp_2>>, 3>& weights_patch2)
    {
        using Grid1D = std::conditional_t<is_on_patch1, GridEdgePerp1, GridEdgePerp2>;
        static_assert(std::is_same_v<Idx1D, Idx<Grid1D>>);

        Extremity constexpr extremity = is_on_patch1 ? s_extremity_1 : s_extremity_2;
        IdxStep<Grid1D> idx_step(extremity == FRONT ? 1 : -1);

        for (int current_n(2); current_n < n_cells; current_n++) {
            // Update coefficients.
            cycle_buffers(coeff_deriv_patch);
            cycle_buffers(coeff_deriv_other_patch);
            cycle_buffers(weights_patch1);
            cycle_buffers(weights_patch2);

            Idx1D const idx_minus = interface_idx + (current_n - 1) * idx_step;
            Idx1D const idx = interface_idx + current_n * idx_step;
            Idx1D const idx_plus = interface_idx + (current_n + 1) * idx_step;

            auto [cell_length_left, cell_length_right] = get_cell_lengths<is_on_patch1>(idx);

            // Local coefficient (alpha_i/beta_i) furthest from the interface
            double const coeff_furthest = is_on_patch1
                                                  ? get_beta(cell_length_left, cell_length_right)
                                                  : get_alpha(cell_length_left, cell_length_right);
            // Local coefficient (alpha_i/beta_i) closest to the interface
            double const coeff_closest = is_on_patch1
                                                 ? get_alpha(cell_length_left, cell_length_right)
                                                 : get_beta(cell_length_left, cell_length_right);
            std::array<double, 3> const gammas_i = get_gammas(cell_length_left, cell_length_right);

            double const denom
                    = 1
                      - coeff_closest * coeff_deriv_patch[n_minus_1] / coeff_deriv_patch[n_minus_2];

            // Set (a_{n},b_{n},c_{n}) ---
            coeff_deriv_patch[n] = coeff_furthest * coeff_deriv_patch[n_minus_1] / denom;
            coeff_deriv_other_patch[n]
                    = (coeff_deriv_other_patch[n_minus_1]
                       - coeff_closest * coeff_deriv_patch[n_minus_1] / coeff_deriv_patch[n_minus_2]
                                 * coeff_deriv_other_patch[n_minus_2])
                      / denom;

            // Patch1: c_{n} = (c_{n-1} - alpha_i * b_{n-1} / b_{n-2} * c_{n-2} + b_{n-1} * gamma_i) / denom;
            // Patch2: c_{n} = (c_{n-1} - beta_i * a_{n-1} / a_{n-2} * c_{n-2} + a_{n-1} * gamma_i) / denom;
            double coeff_closer_deriv
                    = coeff_closest * coeff_deriv_patch[n_minus_1] / coeff_deriv_patch[n_minus_2];
            ddc::host_for_each(m_idx_range_perp_1, [&](IdxEdgePerp1 const& idx_perp) {
                weights_patch1[n](idx_perp)
                        = 1. / denom
                          * (weights_patch1[n_minus_1](idx_perp)
                             - coeff_closer_deriv * weights_patch1[n_minus_2](idx_perp));
            });
            ddc::host_for_each(m_idx_range_perp_2, [&](IdxEdgePerp2 const& idx_perp) {
                weights_patch2[n](idx_perp)
                        = 1. / denom
                          * (weights_patch2[n_minus_1](idx_perp)
                             - coeff_closer_deriv * weights_patch2[n_minus_2](idx_perp));
            });
            double const factor = 1. / denom * coeff_deriv_patch[n_minus_1];
            if constexpr (is_on_patch1) {
                weights_patch1[n](idx_plus) += factor * gammas_i[0];
                weights_patch1[n](idx) += factor * gammas_i[1];
                weights_patch1[n](idx_minus) += factor * gammas_i[2];
            } else {
                weights_patch2[n](idx_plus) += factor * gammas_i[2];
                weights_patch2[n](idx) += factor * gammas_i[1];
                weights_patch2[n](idx_minus) += factor * gammas_i[0];
            }
        }
    }

    template <bool is_on_patch1, class Idx1D>
    void correction_boundary(
            int const n_points,
            Idx1D const interface_idx,
            std::array<double, 3>& coeff_deriv_patch,
            std::array<double, 3>& coeff_deriv_other_patch,
            std::array<host_t<DField<IdxRange1DPerp_1>>, 3>& weights_patch1,
            std::array<host_t<DField<IdxRange1DPerp_2>>, 3>& weights_patch2)
    {
        using Grid1D = std::conditional_t<is_on_patch1, GridEdgePerp1, GridEdgePerp2>;
        static_assert(std::is_same_v<Idx1D, Idx<Grid1D>>);

        // Update coefficients.
        cycle_buffers(coeff_deriv_patch);
        cycle_buffers(coeff_deriv_other_patch);
        cycle_buffers(weights_patch1);
        cycle_buffers(weights_patch2);

        Extremity constexpr extremity = is_on_patch1 ? s_extremity_1 : s_extremity_2;
        IdxStep<Grid1D> idx_step(extremity == FRONT ? 1 : -1);

        int const current_n = n_points - 3;
        Idx1D const idx_minus = interface_idx + (current_n - 1) * idx_step;
        Idx1D const idx = interface_idx + current_n * idx_step;
        Idx1D const idx_plus = interface_idx + (current_n + 2) * idx_step;
        // Additional interpolation point.
        Idx1D const idx_star = interface_idx + (current_n + 1) * idx_step;

        auto [cell_length_left, cell_length_right] = get_cell_lengths<is_on_patch1>(idx);
        auto [cell_length_left_star, cell_length_right_star]
                = get_cell_lengths<is_on_patch1>(idx_star);

        if constexpr (is_on_patch1) {
            cell_length_left += cell_length_left_star;
        } else {
            cell_length_right += cell_length_right_star;
        }

        // Local coefficient (alpha_i/beta_i) furthest from the interface
        double coeff_furthest = is_on_patch1 ? get_beta(cell_length_left, cell_length_right)
                                             : get_alpha(cell_length_left, cell_length_right);
        // Local coefficient (alpha_i/beta_i) closest to the interface
        double coeff_closest = is_on_patch1 ? get_alpha(cell_length_left, cell_length_right)
                                            : get_beta(cell_length_left, cell_length_right);
        std::array<double, 3> const gammas_i = get_gammas(cell_length_left, cell_length_right);

        // Correction of gammas_i, alpha_i and beta_i
        double x;
        if constexpr (is_on_patch1) {
            // (x^-_star - x^-_{i-1}) / Delta x^-_i
            x = cell_length_left_star / cell_length_left;
        } else {
            // (x^+_star - x^+_i) / Delta x^+_i
            x = cell_length_left_star / cell_length_right;
        }
        double const H0 = (1 - x) * (1 - x) * (1 + 2 * x);
        double const H1 = x * x * (3 - 2 * x);
        double const K0 = (1 - x) * (1 - x) * x;
        double const K1 = x * x * (x - 1);

        double gamma_plus;
        double gamma_star;
        double gamma_idx;
        double gamma_minus;
        if constexpr (is_on_patch1) {
            double denom_gamma = 1. / (1 + coeff_furthest * K1 / K0);

            coeff_closest = denom_gamma * coeff_closest;

            gamma_plus = denom_gamma * (gammas_i[0] - coeff_furthest * H0 / K0 / cell_length_left);
            gamma_star = denom_gamma * coeff_furthest / K0 / cell_length_left;
            gamma_idx = denom_gamma * (gammas_i[1] - coeff_furthest * H1 / K0 / cell_length_left);
            gamma_minus = denom_gamma * gammas_i[2];
        } else {
            double denom_gamma = 1. / (1 + coeff_furthest * K0 / K1);

            coeff_closest = denom_gamma * coeff_closest;

            gamma_plus = denom_gamma * (gammas_i[2] - coeff_furthest * H1 / K1 / cell_length_right);
            gamma_star = denom_gamma * coeff_furthest / K1 / cell_length_right;
            gamma_idx = denom_gamma * (gammas_i[1] - coeff_furthest * H0 / K1 / cell_length_right);
            gamma_minus = denom_gamma * gammas_i[0];
        }

        // No more boundary derivative, the local coefficient is set to zero.
        coeff_furthest = 0;

        double denom
                = 1 - coeff_closest * coeff_deriv_patch[n_minus_1] / coeff_deriv_patch[n_minus_2];

        // Set (a_{n},b_{n},c_{n}) ---
        coeff_deriv_patch[n] = coeff_furthest * coeff_deriv_patch[n_minus_1] / denom;
        coeff_deriv_other_patch[n]
                = (coeff_deriv_other_patch[n_minus_1]
                   - coeff_closest * coeff_deriv_patch[n_minus_1] / coeff_deriv_patch[n_minus_2]
                             * coeff_deriv_other_patch[n_minus_2])
                  / denom;

        // Patch1: c_{n} = (c_{n-1} - alpha_i * b_{n-1} / b_{n-2} * c_{n-2} + b_{n-1} * gamma_i) / denom;
        // Patch2: c_{n} = (c_{n-1} - beta_i * a_{n-1} / a_{n-2} * c_{n-2} + a_{n-1} * gamma_i) / denom;
        double coeff_closer_deriv
                = coeff_closest * coeff_deriv_patch[n_minus_1] / coeff_deriv_patch[n_minus_2];
        ddc::host_for_each(m_idx_range_perp_1, [&](IdxEdgePerp1 const& idx_perp) {
            weights_patch1[n](idx_perp)
                    = 1. / denom
                      * (weights_patch1[n_minus_1](idx_perp)
                         - coeff_closer_deriv * weights_patch1[n_minus_2](idx_perp));
        });
        ddc::host_for_each(m_idx_range_perp_2, [&](IdxEdgePerp2 const& idx_perp) {
            weights_patch2[n](idx_perp)
                    = 1. / denom
                      * (weights_patch2[n_minus_1](idx_perp)
                         - coeff_closer_deriv * weights_patch2[n_minus_2](idx_perp));
        });
        double const factor = 1. / denom * coeff_deriv_patch[n_minus_1];
        if constexpr (is_on_patch1) {
            weights_patch1[n](idx_plus) += factor * gamma_plus;
            weights_patch1[n](idx_star) += factor * gamma_star;
            weights_patch1[n](idx) += factor * gamma_idx;
            weights_patch1[n](idx_minus) += factor * gamma_minus;
        } else {
            weights_patch2[n](idx_plus) += factor * gamma_plus;
            weights_patch2[n](idx_star) += factor * gamma_star;
            weights_patch2[n](idx) += factor * gamma_idx;
            weights_patch2[n](idx_minus) += factor * gamma_minus;
        }
    }


    // ===========================================================================================
    // UNIFORM INTERPOLATION POINTS                                                              |
    // ===========================================================================================
    void set_coefficients_uniform_per_patch_case()
    {
        int const n_cells_1 = m_idx_range_perp_1.size() - 1;
        int const n_cells_2 = m_idx_range_perp_2.size() - 1;

        IdxEdgePerp1 interface_idx_1 = get_extremity_idx(s_extremity_1, m_idx_range_perp_1);
        IdxEdgePerp2 interface_idx_2 = get_extremity_idx(s_extremity_2, m_idx_range_perp_2);

        double cell_length_left = ddc::discrete_space<GridEdgePerp1>().step();
        double cell_length_right = ddc::discrete_space<GridEdgePerp2>().step();

        double const a_11 = get_alpha(cell_length_left, cell_length_right);
        double const b_11 = get_beta(cell_length_left, cell_length_right);

        double const u_plus = 2 + Kokkos::sqrt(3);
        double const u_minus = 2 - Kokkos::sqrt(3);

        // Calculate the values u_k that are needed for the coefficients
        double const un1 = Kokkos::pow(u_plus, n_cells_1) - Kokkos::pow(u_minus, n_cells_1);
        double const un1_minus
                = Kokkos::pow(u_plus, n_cells_1 - 1) - Kokkos::pow(u_minus, n_cells_1 - 1);

        double const un2 = Kokkos::pow(u_plus, n_cells_2) - Kokkos::pow(u_minus, n_cells_2);
        double const un2_minus
                = Kokkos::pow(u_plus, n_cells_2 - 1) - Kokkos::pow(u_minus, n_cells_2 - 1);

        // Calculate denominator used for all coefficients
        double const denominator = un1 * un2 + un1 * un2_minus * a_11 + un1_minus * un2 * b_11;


        // Compute the coefficients a and b.
        double const u1 = 2 * Kokkos::sqrt(3);
        m_coeff_deriv_patch_1 = neg_1_pow(n_cells_1 - 1) * u1 * b_11 * un2 / denominator;
        m_coeff_deriv_patch_2 = neg_1_pow(n_cells_2 - 1) * u1 * a_11 * un1 / denominator;


        // Compute the weights {omega_k}_{k = -NL, ..., NR} in c.
        double const factor_a = 3 * a_11 / cell_length_right / denominator;
        double const factor_b = 3 * b_11 / cell_length_left / denominator;

        // --- for k = 0
        m_weights_patch_2(interface_idx_2) = -factor_b * un2 * (un1 - un1_minus);

        // --- for k = 1, ..., NR
        set_local_uniform_weights(m_weights_patch_2, factor_a, un1, interface_idx_2);

        m_weights_patch_1(interface_idx_1) = factor_a * un1 * (un2 - un2_minus);
        set_local_uniform_weights(m_weights_patch_1, -factor_b, un2, interface_idx_1);
    };

    template <class EdgePerpGrid>
    static void set_local_uniform_weights(
            host_t<DField<IdxRange<EdgePerpGrid>>> weights,
            double factor,
            double un_other_patch,
            Idx<EdgePerpGrid> idx_interface)
    {
        double const u_plus = 2 + Kokkos::sqrt(3);
        double const u_minus = 2 - Kokkos::sqrt(3);

        IdxRange<EdgePerpGrid> idx_range = get_idx_range(weights);
        Idx<EdgePerpGrid> idx_outside
                = (idx_interface != idx_range.front()) ? idx_range.front() : idx_range.back();

        int n_cells = idx_range.size() - 1;

        IdxStep<EdgePerpGrid> edge_pt(1);
        IdxRange<EdgePerpGrid> central_idx_range = idx_range.remove(edge_pt, edge_pt);

        ddc::host_for_each(central_idx_range, [&](Idx<EdgePerpGrid> idx) {
            const int k = abs(idx - idx_interface);
            const int n_minus_k = n_cells - k;
            double const vk_minus
                    = Kokkos::pow(u_plus, n_minus_k - 1) - Kokkos::pow(u_minus, n_minus_k - 1);
            double const vk_plus
                    = Kokkos::pow(u_plus, n_minus_k + 1) - Kokkos::pow(u_minus, n_minus_k + 1);

            weights(idx) = neg_1_pow(k) * factor * un_other_patch * (vk_plus - vk_minus);
        });

        weights(idx_outside) = neg_1_pow(n_cells) * factor * un_other_patch * 2 * Kokkos::sqrt(3);
    }

    template <bool is_on_patch1, class Idx1D>
    std::tuple<double, double> get_cell_lengths(Idx1D const& idx) const
    {
        // Tag for the grid on this and the other patch.
        using Grid1D = std::conditional_t<is_on_patch1, GridEdgePerp1, GridEdgePerp2>;
        using OGrid1D = std::conditional_t<is_on_patch1, GridEdgePerp2, GridEdgePerp1>;
        static_assert(std::is_same_v<Idx1D, Idx<Grid1D>>);

        Extremity constexpr extremity = is_on_patch1 ? s_extremity_1 : s_extremity_2;
        Extremity constexpr other_extremity = is_on_patch1 ? s_extremity_2 : s_extremity_1;

        IdxRange<Grid1D> idx_range_1d;
        IdxRange<OGrid1D> other_idx_range_1d;
        if constexpr (is_on_patch1) {
            idx_range_1d = m_idx_range_perp_1;
            other_idx_range_1d = m_idx_range_perp_2;
        } else {
            idx_range_1d = m_idx_range_perp_2;
            other_idx_range_1d = m_idx_range_perp_1;
        }

        // Please, do not provide a index on the boundary except for the interface.
        assert(!(idx == idx_range_1d.front() && extremity == BACK));
        assert(!(idx == idx_range_1d.back() && extremity == FRONT));

        double length_left;
        double length_right;

        Idx<Grid1D> const idx_minus = idx - IdxStep<Grid1D>(1);
        Idx<Grid1D> const idx_plus = idx + IdxStep<Grid1D>(1);

        bool constexpr is_same_orientation_as_global = (is_on_patch1 && s_extremity_1 == BACK)
                                                       || (!is_on_patch1 && s_extremity_2 == FRONT);

        // If the given index corresponds to the index at the interface,
        Idx<Grid1D> const idx_interface = get_extremity_idx(extremity, idx_range_1d);
        if (idx == idx_interface) {
            IdxStep<Grid1D> idx_step(extremity == FRONT ? 1 : -1);
            Idx<Grid1D> const idx_incremented = idx + idx_step;

            IdxStep<OGrid1D> other_idx_step(other_extremity == FRONT ? 1 : -1);
            Idx<OGrid1D> const other_idx = get_extremity_idx(other_extremity, other_idx_range_1d);
            Idx<OGrid1D> const other_idx_incremented = other_idx + other_idx_step;

            if constexpr (std::is_same_v<Grid1D, GridEdgePerp2>) {
                length_left
                        = abs(ddc::coordinate(other_idx_incremented) - ddc::coordinate(other_idx));
                length_right = abs(ddc::coordinate(idx_incremented) - ddc::coordinate(idx));
            } else {
                length_left = abs(ddc::coordinate(idx_incremented) - ddc::coordinate(idx));
                length_right
                        = abs(ddc::coordinate(other_idx_incremented) - ddc::coordinate(other_idx));
            }
        }
        // If given index is on the patch 1 or on the patch 2,
        else if (is_same_orientation_as_global) {
            // . | →   or  → | .
            length_left = abs(ddc::coordinate(idx) - ddc::coordinate(idx_minus));
            length_right = abs(ddc::coordinate(idx_plus) - ddc::coordinate(idx));
        } else {
            //  . | ←  or  ← | .
            length_left = abs(ddc::coordinate(idx_plus) - ddc::coordinate(idx));
            length_right = abs(ddc::coordinate(idx) - ddc::coordinate(idx_minus));
        }

        if ((length_left == 0) || (length_right == 0)) {
            throw std::runtime_error(
                    "[abort] Ill-defined: the length of the cells must be not zero.");
        }
        return std::tuple<double, double>(length_left, length_right);
    }


    std::array<double, 3> get_gammas(double const cell_length_left, double const cell_length_right)
            const
    {
        double const factor = 3. / 2. / (cell_length_left + cell_length_right);
        double const R_on_L = cell_length_right / cell_length_left;
        double const L_on_R = cell_length_left / cell_length_right;

        double gamma_i_minus = -factor * R_on_L;
        double gamma_i = factor * (R_on_L - L_on_R);
        double gamma_i_plus = factor * L_on_R;

        return std::array<double, 3>({gamma_i_minus, gamma_i, gamma_i_plus});
    }

    static double get_alpha(double const cell_length_left, double const cell_length_right)
    {
        return -0.5 * cell_length_left / (cell_length_right + cell_length_left);
    }

    static double get_beta(double const cell_length_left, double const cell_length_right)
    {
        return -0.5 * cell_length_right / (cell_length_right + cell_length_left);
    }


    template <class Grid1D>
    KOKKOS_INLINE_FUNCTION Idx<Grid1D> get_extremity_idx(
            Extremity const extremity,
            IdxRange<Grid1D> const& idx_range) const
    {
        return (extremity == FRONT) ? idx_range.front() : idx_range.back();
    }

    static void cycle_buffers(std::array<double, 3>& buffer)
    {
        // n-1 becomes n-2
        buffer[n_minus_2] = buffer[n_minus_1];
        // n becomes n-1
        buffer[n_minus_1] = buffer[n];
        // new n is ready to be defined (but still contains value of old n)
    }

    template <class IdxRangeType, class MemorySpace>
    static void cycle_buffers(std::array<DField<IdxRangeType, MemorySpace>, 3>& buffer)
    {
        // Save pointer to n-2 memory block, whose values are no longer needed
        DField<IdxRangeType, MemorySpace> tmp = buffer[0];
        // n-1 becomes n-2
        buffer[n_minus_2] = buffer[n_minus_1];
        // n becomes n-1
        buffer[n_minus_1] = buffer[n];
        // new n is ready to be defined (but still contains values of old n-2)
        buffer[n] = tmp;
    }

    template <class IdxRangeType>
    static std::array<DField<IdxRangeType, Kokkos::HostSpace>, 3> build_array(
            int input_position,
            DField<IdxRangeType, Kokkos::HostSpace> persistent_memory,
            DField<IdxRangeType, Kokkos::HostSpace> buffer_1,
            DField<IdxRangeType, Kokkos::HostSpace> buffer_2)
    {
        if (input_position == 0) {
            return {persistent_memory, buffer_1, buffer_2};
        } else if (input_position == 1) {
            return {buffer_1, persistent_memory, buffer_2};
        } else {
            return {buffer_1, buffer_2, persistent_memory};
        }
    }
